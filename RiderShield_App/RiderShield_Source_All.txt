class RiderScope extends InheritedNotifier<RiderState> {
  const RiderScope({super.key, required RiderState state, required super.child}) : super(notifier: state);

  static RiderState of(BuildContext context) {
    final scope = context.dependOnInheritedWidgetOfExactType<RiderScope>();
    assert(scope != null, 'RiderScope missing above the widget tree');
    return scope!.notifier!;
  }
}

class CrashIncident {
  const CrashIncident({
    required this.id,
    required this.happenedAt,
    required this.location,
    required this.severity,
    required this.sosSent,
    this.notificationStatus = 'notSent',
    this.testOnly = true,
  });

  final String id;
  final DateTime happenedAt;
  final String location;
  final String severity;
  final bool sosSent;
  final String notificationStatus;
  final bool testOnly;

  Map<String, dynamic> toJson() => {
        'id': id,
        'happenedAt': happenedAt.toIso8601String(),
        'location': location,
        'severity': severity,
        'sosSent': sosSent,
        'notificationStatus': notificationStatus,
        'testOnly': testOnly,
      };

  factory CrashIncident.fromJson(
    Map<String, dynamic> j,
  ) =>
      CrashIncident(
        id: '${j['id'] ?? ''}',
        happenedAt: DateTime.tryParse(
              '${j['happenedAt'] ?? ''}',
            ) ??
            DateTime.now(),
        location: '${j['location'] ?? 'Not recorded'}',
        severity: '${j['severity'] ?? 'Unknown'}',
        sosSent: j['sosSent'] == true,
        notificationStatus:
            '${j['notificationStatus'] ?? 'notSent'}',
        testOnly: j['testOnly'] != false,
      );
}

  speed,
  drowsiness,
  vehicleProximity,

class WarningEvent {
  const WarningEvent({
    required this.id,
    required this.happenedAt,
    required this.type,
    this.detail = '',
    this.speedLevel,
    this.drowsinessLevel,
    this.proximityLevel,
  });

  final String id;
  final DateTime happenedAt;
  final WarningType type;
  final String detail;

  final String? speedLevel;
  final String? drowsinessLevel;
  final String? proximityLevel;

  Map<String, dynamic> toJson() => {
        'id': id,
        'happenedAt': happenedAt.toIso8601String(),
        'type': type.name,
        'detail': detail,
        'speedLevel': speedLevel,
        'drowsinessLevel': drowsinessLevel,
        'proximityLevel': proximityLevel,
      };

  factory WarningEvent.fromJson(
    Map<String, dynamic> j,
  ) =>
      WarningEvent(
        id: '${j['id'] ?? ''}',
        happenedAt: DateTime.tryParse(
              '${j['happenedAt'] ?? ''}',
            ) ??
            DateTime.now(),
        type: WarningType.values.firstWhere(
          (t) => t.name == j['type'],
          orElse: () => WarningType.speed,
        ),
        detail: '${j['detail'] ?? ''}',
        speedLevel: j['speedLevel'] as String?,
        drowsinessLevel:
            j['drowsinessLevel'] as String?,
        proximityLevel:
            j['proximityLevel'] as String?,
      );
}

        prefs.getString('rs_avatar') ?? '';

        prefs.getString('rs_cover') ?? '';

    if (savedDelay != null) {
      SosSettings.delaySeconds.value =
          _clampDelay(savedDelay);
    }

    SosSettings.delaySeconds.addListener(
      s._onSosDelayChanged,
    );

    try {
      final raw = jsonDecode(
        prefs.getString('rs_contacts') ?? '[]',

      s.incidents.sort(
        (a, b) => b.happenedAt.compareTo(a.happenedAt),

        prefs.getString('rs_warnings') ?? '[]',

        raw.map(

        (a, b) => b.happenedAt.compareTo(a.happenedAt),

  Future<void> setLoginDetails({
    String? name,
    String? emailAddress,
  }) async {
    final newEmail = (emailAddress ?? '').trim();
    final newName = (name ?? '').trim();

    final switchingAccount =
        newEmail.isNotEmpty &&
        newEmail.toLowerCase() !=
            email.trim().toLowerCase();

    if (switchingAccount) {
      // Save previous account.
      _saveSnapshot();

      // Clear previous account.
      _blank();

      // Restore saved data for new account.
      _loadSnapshot(newEmail);
    }

    if (newEmail.isNotEmpty) {
      email = newEmail;
    }

    // Never overwrite a saved real name with an
    // empty value from the sign-in screen.
    if (newName.isNotEmpty) {
      fullName = newName;
    }

    _ensureSampleProfile();

    // Persist after all values are updated.
    await _persistLive();

    _saveSnapshot();

    notifyListeners();
  }

  Future<void> upsertContact(
    EmergencyContact value,
  ) async {
    final index = contacts.indexWhere(
      (c) => c.id == value.id,
    );

    if (index == -1) {
      contacts.add(value);
    } else {
      contacts[index] = value;
    }

    notifyListeners();

    await _prefs.setString(
      'rs_contacts',
      jsonEncode(
        contacts.map((c) => c.toJson()).toList(),
      ),
    );

    _saveSnapshot();
  }

  Future<void> addWarning(WarningEvent w) async {
    warnings.insert(0, w);

    warnings.sort(
      (a, b) => b.happenedAt.compareTo(a.happenedAt),
    );

    if (warnings.length > _maxWarnings) {
      warnings.removeRange(
        _maxWarnings,
        warnings.length,
      );
    }

    notifyListeners();

    await _saveWarnings();
    _saveSnapshot();
  }

  // ====================================================

  bool _suspendSosSave = false;

  static int _clampDelay(int v) {
    if (v < SosSettings.minDelay) {
      return SosSettings.minDelay;
    }

    if (v > SosSettings.maxDelay) {
      return SosSettings.maxDelay;
    }

    return v;
  }

  void _setDelay(int v) {
    _suspendSosSave = true;

    SosSettings.delaySeconds.value = v;

    _suspendSosSave = false;
  }

  void _onSosDelayChanged() {
    if (_suspendSosSave) return;

    _prefs.setInt(
      'rs_sos_delay',
      SosSettings.delaySeconds.value,
    );

    _saveSnapshot();
  }

  void _loadSnapshot(String e) {
    final raw = _prefs.getString(_accKey(e));

    if (raw == null) return;

    try {
      final j = Map<String, dynamic>.from(
        jsonDecode(raw) as Map,
      );

      fullName = '${j['name'] ?? ''}';
      email = '${j['email'] ?? ''}';
      phone = '${j['phone'] ?? ''}';
      city = '${j['city'] ?? ''}';

      ridingExperience =
          '${j['experience'] ?? 'Beginner'}';

      avatarBase64 = '${j['avatar'] ?? ''}';
      coverBase64 = '${j['cover'] ?? ''}';

      final d = j['sosDelay'];

      _setDelay(
        _clampDelay(d is int ? d : 10),
      );

      contacts.addAll(
        ((j['contacts'] as List?) ?? []).map(
          (x) => EmergencyContact.fromJson(
            Map<String, dynamic>.from(x as Map),
          ),
        ),
      );

      incidents.addAll(
        ((j['incidents'] as List?) ?? []).map(
          (x) => CrashIncident.fromJson(
            Map<String, dynamic>.from(x as Map),
          ),
        ),
      );

      warnings.addAll(
        ((j['warnings'] as List?) ?? []).map(
          (x) => WarningEvent.fromJson(
            Map<String, dynamic>.from(x as Map),
          ),
        ),
      );

      incidents.sort(
        (a, b) => b.happenedAt.compareTo(a.happenedAt),
      );

      warnings.sort(
        (a, b) => b.happenedAt.compareTo(a.happenedAt),
      );
    } catch (_) {
      // Corrupted account snapshot.
    }
  }

      _prefs.setString('rs_phone', phone),
      _prefs.setString('rs_city', city),
      _prefs.setString(

  }

  );

/// Real crash detectors should open CrashAlertScreen(crashDetected: true,
/// testOnly: false). This UI does not itself detect crashes. When testOnly is
/// true, notifications are categorically blocked by EmergencyDispatcher.
class CrashAlertScreen extends StatefulWidget {
  const CrashAlertScreen({super.key,
    required this.crashDetected, this.testOnly = false,
    this.manualCountdown = false});
  final bool crashDetected;
  final bool testOnly;
  /// User pressed SOS: run the countdown, then send a manual SOS (no crash report).
  final bool manualCountdown;

  @override
  State<CrashAlertScreen> createState() => _CrashAlertScreenState();
}

class _CrashAlertScreenState extends State<CrashAlertScreen>
    with WidgetsBindingObserver {
  Timer? _timer;
  DateTime? _deadline;
  // Countdown length chosen by the user on the Home screen (read once when the alert opens).
  late final int _totalSeconds = SosSettings.delaySeconds.value;
  late int _secondsLeft = _totalSeconds;
  bool _processing = false;
  bool _finished = false;
  bool _canLeave = false;
  EmergencyOutcome? _result;

  bool get _hasCountdown => widget.crashDetected || widget.manualCountdown;

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    if (_hasCountdown) {
      _deadline = DateTime.now().add(Duration(seconds: _totalSeconds));
      _timer = Timer.periodic(const Duration(milliseconds: 150), (_) => _tick());
    }
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.resumed) _tick();
  }

  void _tick() {
    if (_processing || _finished || _deadline == null) return;
    final ms = _deadline!.difference(DateTime.now()).inMilliseconds;
    final left = ms <= 0 ? 0 : (ms / 1000).ceil();
    if (left != _secondsLeft && mounted) setState(() => _secondsLeft = left);
    if (ms <= 0) _send();
  }

  void _cancel() {
    if (_processing || _finished) return;
    _timer?.cancel();
    setState(() => _canLeave = true);
    WidgetsBinding.instance.addPostFrameCallback((_) {
      if (mounted) Navigator.of(context).pop();
    });
  }

  Future<void> _send() async {
    if (_processing || _finished || !mounted) return;
    _timer?.cancel();
    setState(() => _processing = true);
    final rider = RiderScope.of(context);
    final result = await EmergencyDispatcher.dispatch(
      rider: rider,
      simulated: widget.testOnly,
      trigger: widget.crashDetected ? 'crash_timeout' : 'manual_sos',
    );
    if (widget.crashDetected) {
      // Real/crash-test records ONLY. Manual SOS and warnings don't create reports.
      await rider.recordCrash(
        location: result.location,
        notificationStatus: result.status.name,
        testOnly: widget.testOnly,
      );
    }
    if (!mounted) return;
    setState(() { _result = result; _finished = true; _processing = false; });
  }

  Future<void> _openSms(String phone) async {
    final rider = RiderScope.of(context);
    var place = _result?.location ?? 'Location unavailable';
    if (place.startsWith('Location not requested')) {
      try {
        place = await EmergencyDispatcher.tryLocation();
      } catch (_) {
        place = 'Location unavailable';
      }
    }
    final message = 'RiderShield emergency alert. '
      '${rider.fullName.isEmpty ? 'A rider' : rider.fullName} may need help. '
      'Time: ${DateTime.now().toLocal()}. Location: $place. '
      'Please try to contact them and reach emergency services if needed.';

    // Keep only digits and a leading "+", so spaces or brackets in the saved
    // number can't break the sms: link. encodeComponent keeps spaces as %20
    // (queryParameters would turn them into "+" signs in the message).
    final number = phone.replaceAll(RegExp(r'[^0-9+]'), '');
    final uri = Uri.parse('sms:$number?body=${Uri.encodeComponent(message)}');

    var opened = false;
    try {
      opened = await launchUrl(uri, mode: LaunchMode.externalApplication);
    } catch (_) {
      opened = false; // e.g. Chrome/desktop with no Messages app
    }

    if (!opened) {
      try {
        await Clipboard.setData(ClipboardData(text: message));
      } catch (_) {/* clipboard not available */}
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(const SnackBar(
          content: Text('Messages unavailable. SOS text copied; no SMS sent.'),
        ));
      }
    }
  }

class _DashboardShellState extends State<DashboardShell> {
  int currentTab = 0;

  int _previousTab = 0;
  int _previousProfileTab = 0;

  void openTab(int tab) {
    if (tab < 0 || tab > 4 || tab == currentTab) return;

    setState(() {
      if (tab == 1 && currentTab != 1) {
        _previousTab = currentTab;
      }

      if (tab == 4 && currentTab != 4) {
        _previousProfileTab = currentTab;
      }

      currentTab = tab;
    });
  }

  void _backFromReports() {
    openTab(_previousTab == 1 ? 0 : _previousTab);
  }

  void _backFromProfile() {
    openTab(
      _previousProfileTab == 4
          ? 0
          : _previousProfileTab,
    );
  }

  // ------------------------------------------
  // HEADER
  // ------------------------------------------

  Widget _buildHeader() {
    if (currentTab == 4) {
      return _backHeader(
        title: 'Profile',
        onBack: _backFromProfile,
      );
    }

    if (currentTab == 1) {
      return _backHeader(
        title: 'Safety Report',
        onBack: _backFromReports,
      );
    }

    if (currentTab == 2) {
      return _simpleHeader('Emergency');
    }

    if (currentTab == 3) {
      return _simpleHeader('Connect Helmet');
    }

    // HOME
    // Rebuild when RiderState changes.
    final state = RiderScope.of(context);

    return ListenableBuilder(
      listenable: state,
      builder: (context, _) {
        return _HomeHeader(
          name: state.shortName,

  Future<void> _startSos(
    BuildContext context,
    List<EmergencyContact> contacts,
  ) async {
    if (contacts.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          behavior: SnackBarBehavior.floating,
          content: Text(
            'Add an emergency contact first.',
          ),
        ),
      );

      openTab(_emergencyTab);
      return;
    }

    final sendToAll = await showDialog<bool>(
      context: context,
      barrierDismissible: false,
      builder: (_) => _SosConfirmDialog(
        count: contacts.length,
      ),
    );

    if (sendToAll == null || !context.mounted) {
      return;
    }

    if (sendToAll) {
      _sendSos(context, contacts);
      return;
    }

    final chosen =
        await showModalBottomSheet<List<EmergencyContact>>(
      context: context,
      isScrollControlled: true,
      showDragHandle: true,
      backgroundColor: const Color(0xFF173550),
      builder: (_) => _ContactPicker(
        contacts: contacts,
      ),
    );

    if (chosen == null ||
        chosen.isEmpty ||
        !context.mounted) {
      return;
    }

    _sendSos(context, chosen);
  }

    BuildContext context,

          'SOS request confirmed for ${to.length} contact(s): $names. Messages have not been sent.',

      const Duration(seconds: 1),

          const SizedBox(height: 14),
          ClipRRect(
            borderRadius: BorderRadius.circular(6),
            child: LinearProgressIndicator(
              value: _left / _confirmSeconds,
              minHeight: 6,
              color: const Color(0xFFE5484D),
              backgroundColor: Colors.white12,
            ),
          ),
          const SizedBox(height: 8),
          Text(

    newPassword.dispose();
    confirmPassword.dispose();
    super.dispose();
  }

  String? validatePassword(String? value) {
    final p = value ?? '';

    if (p.length < 8) {
      return 'Use at least 8 characters';
    }
    if (!RegExp(r'[A-Z]').hasMatch(p)) {
      return 'Add an uppercase letter';
    }
    if (!RegExp(r'[a-z]').hasMatch(p)) {
      return 'Add a lowercase letter';
    }
    if (!RegExp(r'[0-9]').hasMatch(p)) {
      return 'Add a number';
    }
    if (!RegExp(r'[^A-Za-z0-9]').hasMatch(p)) {
      return 'Add a symbol';
    }

    return null;

  DateTimeRange? _crashDateRange;
  int? _recentDays; // null = All, -1 = Custom

  bool _isInCrashDateRange(CrashIncident incident) {
    if (_recentDays == null) return true;
    if (_recentDays != -1) {
      final today = DateUtils.dateOnly(DateTime.now());
      final start = today.subtract(Duration(days: _recentDays! - 1));
      final end = today.add(const Duration(days: 1));
      return !incident.happenedAt.isBefore(start) &&
          incident.happenedAt.isBefore(end);
    }
    final range = _crashDateRange;
    if (range == null) return true;

    final date = incident.happenedAt;
    final start = DateUtils.dateOnly(range.start);
    final endExclusive =
        DateUtils.dateOnly(range.end).add(const Duration(days: 1));

    return !date.isBefore(start) && date.isBefore(endExclusive);
  }

      ),

        builder: (sheetContext, updateSheet) {
          Widget option(String label, int? days) {
            final selected = draftDays == days;

  String _formatRangeDate(DateTime date) {
    const months = [
      'Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun',
      'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec',
    ];

    return '${date.day} ${months[date.month - 1]} ${date.year}';
  }

  // Location helpers
  bool _isMapLink(String location) => location.startsWith('http');

  String _placeLabel(String location) {
    if (!_isMapLink(location)) return location;
    final q = Uri.tryParse(location)?.queryParameters['q'];
    return q == null ? 'GPS location' : 'GPS: ${q.replaceAll(',', ', ')}';
  }

  Future<void> _openMap(BuildContext context, String link) async {
    final ok = await launchUrl(
      Uri.parse(link),
      mode: LaunchMode.externalApplication,
    );

    if (!ok && context.mounted) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Could not open Maps.')),
      );
    }
  }

  // Crashes are classified by the SOS emergency request.
  bool _sosAccepted(CrashIncident incident) {
    final status = incident.notificationStatus
        .toLowerCase()
        .replaceAll('_', ' ')
        .trim();
    return status.startsWith('accept');
  }

  String _sosLabel(bool accepted) =>
      accepted ? 'SOS sent' : 'SOS not sent';

  Color _sosColor(bool accepted) => accepted
      ? const Color(0xFFFF6B72)
      : const Color(0xFF8ED5FF);

  String _statusLabel(String raw) {
    if (raw.isEmpty) return 'Unknown';

    final spaced = raw
        .replaceAllMapped(
          RegExp(r'([a-z])([A-Z])'),
          (m) => '${m[1]} ${m[2]}',
        )
        .replaceAll('_', ' ');

    return spaced[0].toUpperCase() + spaced.substring(1);
  }

  // ---------- Warning level ----------

  (String?, String?, String?) _factors(WarningEvent w) {
    if (w.speedLevel == null &&
        w.drowsinessLevel == null &&
        w.proximityLevel == null) {
      return (
        w.type == WarningType.speed ? 'Low' : null,
        w.type == WarningType.drowsiness ? 'Low' : null,
        w.type == WarningType.vehicleProximity ? 'Low' : null,
      );
    }
    return (w.speedLevel, w.drowsinessLevel, w.proximityLevel);
  }

  int _rank(String? level) => switch (level) {
        'High' => 3,
        'Medium' => 2,
        'Low' => 1,
        _ => 0,
      };

  String _warningLevel(WarningEvent w) {
    final (s, d, p) = _factors(w);
    final top = [s, d, p].map(_rank).reduce((a, b) => a > b ? a : b);
    return switch (top) {
      3 => 'High',
      2 => 'Medium',
      _ => 'Low',
    };
  }

  // Only High and Medium warnings are shown.
  List<WarningEvent> _shownWarnings(RiderState state) =>
      state.warnings.where((w) => _warningLevel(w) != 'Low').toList();

  Color _warningLevelColor(String level) => switch (level) {
        'High' => const Color(0xFFFF6B72),
        'Medium' => const Color(0xFFFFB45C),
        _ => const Color(0xFF7BE0AE),
      };

  void showWarning(BuildContext context, WarningEvent w) {

class _SignUpScreenState extends State<SignUpScreen> {
  final formKey = GlobalKey<FormState>();

  final name = TextEditingController();
  final email = TextEditingController();
  final password = TextEditingController();
  final confirmation = TextEditingController();

  bool canSubmit = false;

  @override
  void initState() {
    super.initState();

    name.addListener(updateFormStatus);
    email.addListener(updateFormStatus);
    password.addListener(updateFormStatus);
    confirmation.addListener(updateFormStatus);
  }

  String? checkName(String? value) {
    final text = (value ?? '').trim();

    if (text.isEmpty) {
      return 'Full name is required';
    }

    if (text.length < 2) {
      return 'Enter at least 2 characters';
    }

    return null;
  }

  String? checkPassword(String? value) {
    final text = value ?? '';

    if (text.isEmpty) {
      return 'Password is required';
    }

    if (text.length < 8) {
      return 'Use at least 8 characters';
    }

    if (!RegExp(r'[A-Z]').hasMatch(text)) {
      return 'Add an uppercase letter (A-Z)';
    }

    if (!RegExp(r'[a-z]').hasMatch(text)) {
      return 'Add a lowercase letter (a-z)';
    }

    if (!RegExp(r'[0-9]').hasMatch(text)) {
      return 'Add at least one number';
    }

    if (!RegExp(r'[^A-Za-z0-9]').hasMatch(text)) {
      return 'Add at least one special character';
    }

    return null;
  }

  String? checkConfirmation(String? value) {
    final text = value ?? '';

    if (text.isEmpty) {
      return 'Please confirm your password';
    }

    if (text != password.text) {
      return 'Passwords do not match';
    }

    return null;
  }

  void updateFormStatus() {
    if (!mounted) return;

    final valid =
        checkName(name.text) == null &&
        validateEmail(email.text) == null &&
        checkPassword(password.text) == null &&
        checkConfirmation(confirmation.text) == null;

    if (canSubmit != valid) {
      setState(() {
        canSubmit = valid;
      });
    }
  }

  void submit() {
    if (!(formKey.currentState?.validate() ?? false)) {
      return;
    }

    if (!canSubmit) {
      return;

    FocusScope.of(context).unfocus();

    // Current local account flow.

      name: name.text.trim(),
      emailAddress: email.text.trim(),
    );

      rideRoute(const DashboardShell()),
      (route) => false,
    );
  }

  }

import 'dart:convert';
import 'package:geolocator/geolocator.dart';
import 'package:http/http.dart' as http;
import '../app/app_state.dart';

enum DispatchState { demoOnly, noContacts, noService, accepted, failed }

class EmergencyOutcome {
  const EmergencyOutcome({required this.status, required this.message,
    required this.location, required this.sentAt});
  final DispatchState status;
  final String message;
  final String location;
  final DateTime sentAt;
  bool get accepted => status == DispatchState.accepted;
}

/// Implement this URL with an authenticated HTTPS backend that sends SMS
/// to recipients and responds with {"accepted": true} only after the job
/// is queued durably. Never put SMS provider credentials in this mobile app.
/// A 2xx response alone does not mean a message was delivered.
class EmergencyDispatcher {
  EmergencyDispatcher._();
  static const _apiUrl = String.fromEnvironment('RIDERSHIELD_EMERGENCY_API_URL');
  static bool get configured => _apiUrl.startsWith('https://');

  /// Returns a Google Maps link (https://maps.google.com/?q=lat,lng) when the
  /// position is available. The Safety Report opens this link in Maps.
  static Future<String> tryLocation() async {
    try {
      if (!await Geolocator.isLocationServiceEnabled()) return 'Location unavailable (GPS off)';
      var permission = await Geolocator.checkPermission();
      if (permission == LocationPermission.denied) {
        permission = await Geolocator.requestPermission();
      }
      if (permission == LocationPermission.denied ||
          permission == LocationPermission.deniedForever) {
        return 'Location unavailable (permission denied)';
      }
      final position = await Geolocator.getCurrentPosition(
        locationSettings: const LocationSettings(
          accuracy: LocationAccuracy.high,
          timeLimit: Duration(seconds: 7),
        ),
      ).timeout(const Duration(seconds: 8));
      return 'https://maps.google.com/?q=${position.latitude},${position.longitude}';
    } catch (_) {
      return 'Location unavailable';
    }
  }

  static Future<EmergencyOutcome> dispatch({
    required RiderState rider,
    required bool simulated,
    required String trigger,
  }) async {
    final sentAt = DateTime.now();
    // Never query permissions or transmit anything during a test run.
    if (simulated) {
      return EmergencyOutcome(status: DispatchState.demoOnly,
        message: 'TEST ONLY â€” no message was sent.',
        location: 'Test location unavailable', sentAt: sentAt);
    }
    // Real crash: always try to capture the exact location so the crash
    // record in the Safety Report can open it in Google Maps, even when no
    // message can be sent.
    final location = await tryLocation();
    if (rider.contacts.isEmpty) {
      return EmergencyOutcome(status: DispatchState.noContacts,
        message: 'No emergency contacts saved. Nothing was sent.',
        location: location, sentAt: sentAt);
    }
    if (!configured) {
      return EmergencyOutcome(status: DispatchState.noService,
        message: 'Emergency delivery service is not configured. Nothing was sent.',
        location: location, sentAt: sentAt);
    }
    try {
      final response = await http.post(
        Uri.parse(_apiUrl),
        headers: {'content-type': 'application/json'},
        body: jsonEncode({
          'eventId': 'rs-${sentAt.microsecondsSinceEpoch}',
          'trigger': trigger,
          'person': {'name': rider.fullName, 'phone': rider.phone},
          'occurredAt': sentAt.toUtc().toIso8601String(),
          'location': location,
          'recipients': [
            for (final person in rider.contacts)
              {'name': person.name, 'phone': person.phone,
                'relationship': person.relationship},
          ],
        }),
      ).timeout(const Duration(seconds: 12));
      bool accepted = false;
      if (response.statusCode >= 200 && response.statusCode < 300) {
        try {
          final body = jsonDecode(response.body);
          accepted = body is Map && body['accepted'] == true;
        } catch (_) { /* Strict success acknowledgement required. */ }
      }
      return EmergencyOutcome(
        status: accepted ? DispatchState.accepted : DispatchState.failed,
        message: accepted
          ? 'Emergency request accepted by the server; SMS delivery is not yet confirmed.'
          : 'Emergency request was not acknowledged. Check your contacts or call emergency services.',
        location: location, sentAt: sentAt,
      );
    } catch (_) {
      return EmergencyOutcome(status: DispatchState.failed,
        message: 'Could not reach the emergency server. No delivery confirmed.',
        location: location, sentAt: sentAt);
    }
  }
}

import 'package:flutter/material.dart';

// UX-only estimate, not a cryptographic password-security audit.
enum PasswordLevel { empty, weak, medium, strong }

class PasswordStrength {
  const PasswordStrength({
    required this.level,
    required this.tip,
  });

  final PasswordLevel level;
  final String tip;

  String get label => switch (level) {
        PasswordLevel.empty => 'Not entered',
        PasswordLevel.weak => 'Weak',
        PasswordLevel.medium => 'Medium',
        PasswordLevel.strong => 'Strong',
      };

  int get bars => switch (level) {
        PasswordLevel.empty => 0,
        PasswordLevel.weak => 1,
        PasswordLevel.medium => 2,
        PasswordLevel.strong => 3,
      };

  Color get color => switch (level) {
        PasswordLevel.empty => const Color(0xFF698199),
        PasswordLevel.weak => const Color(0xFFFF7F8A),
        PasswordLevel.medium => const Color(0xFFFFC476),
        PasswordLevel.strong => const Color(0xFF78DDC8),
      };
}

PasswordStrength evaluatePassword(String password) {
  if (password.isEmpty) {
    return const PasswordStrength(
      level: PasswordLevel.empty,
      tip: 'Use 8+ characters with uppercase, lowercase, numbers and symbols.',
    );
  }

  final hasLower = RegExp(r'[a-z]').hasMatch(password);
  final hasUpper = RegExp(r'[A-Z]').hasMatch(password);
  final hasNumber = RegExp(r'[0-9]').hasMatch(password);
  final hasSymbol = RegExp(r'[^A-Za-z0-9]').hasMatch(password);

  final categories = [
    hasLower,
    hasUpper,
    hasNumber,
    hasSymbol,
  ].where((value) => value).length;

  final normalized = password.toLowerCase();

  final commonlyUsed = [
    'password',
    'qwerty',
    '12345678',
    'letmein',
    'admin123',
  ].any(normalized.contains);

  // Give accurate advice for the actual missing requirement.
  if (password.length < 8) {
    return const PasswordStrength(
      level: PasswordLevel.weak,
      tip: 'Use at least 8 characters.',
    );
  }

  if (commonlyUsed) {
    return const PasswordStrength(
      level: PasswordLevel.weak,
      tip: 'Avoid common passwords and predictable patterns.',
    );
  }

  if (categories < 2) {
    return const PasswordStrength(
      level: PasswordLevel.weak,
      tip: 'Mix uppercase, lowercase, numbers and symbols.',
    );
  }

  // Match the mandatory sign-up requirements.
  final meetsRequirements =
      hasLower && hasUpper && hasNumber && hasSymbol;

  if (!meetsRequirements) {
    final missing = <String>[];

    if (!hasUpper) missing.add('uppercase letters');
    if (!hasLower) missing.add('lowercase letters');
    if (!hasNumber) missing.add('numbers');
    if (!hasSymbol) missing.add('symbols');

    return PasswordStrength(
      level: PasswordLevel.medium,
      tip: 'Add ${missing.join(', ')}.',
    );
  }

  // All sign-up requirements are satisfied.
  // Extra length improves the strength estimate.
  if (password.length >= 12) {
    return const PasswordStrength(
      level: PasswordLevel.strong,
      tip: 'Great! Your password meets all requirements.',
    );
  }

  return const PasswordStrength(
    level: PasswordLevel.medium,
    tip: 'All requirements met. Use 12+ characters for a stronger password.',
  );
}

// --------------------------------------------------
// NAVBAR GEOMETRY
// --------------------------------------------------

Path _createBaseBarPath(Size size) {
  const double barTop = 37;
  const double cornerRadius = 30;

  return Path()
    ..addRRect(
      RRect.fromRectAndRadius(
        Rect.fromLTWH(
          0,
          barTop,
          size.width,
          size.height - barTop,
        ),
        const Radius.circular(cornerRadius),
      ),
    );
}

Path _createNotchPath(double centerX) {
  return Path()
    ..addOval(
      Rect.fromCircle(
        center: Offset(centerX, 38),
        radius: 40,
      ),
    );
}

Path _createNavbarPath(Size size, double centerX) {
  final basePath = _createBaseBarPath(size);
  final notchPath = _createNotchPath(centerX);

  return Path.combine(
    PathOperation.difference,
    basePath,
    notchPath,
  );
}

// --------------------------------------------------
// CUSTOM NAVBAR PAINTER
// --------------------------------------------------

class _NavbarPainter extends CustomPainter {
  const _NavbarPainter({
    required this.centerX,
  });

  final double centerX;

  @override
  void paint(Canvas canvas, Size size) {
    final basePath = _createBaseBarPath(size);
    final notchPath = _createNotchPath(centerX);
    final barPath = _createNavbarPath(size, centerX);

    // 1. Paint only the bar silhouette.
    // No full-screen rectangular background.
    canvas.drawPath(
      basePath,
      Paint()
        ..color = const Color(0xFF1A1D2B)
        ..style = PaintingStyle.fill
        ..isAntiAlias = true,
    );

    // 2. Fill the moving notch with opaque dark navy.
    // Clipping keeps it strictly inside the bar.
    canvas.save();
    canvas.clipPath(basePath);

    canvas.drawPath(
      notchPath,
      Paint()
        ..color = const Color(0xFF121523)
        ..style = PaintingStyle.fill
        ..isAntiAlias = true,
    );

    canvas.restore();

    // 3. Draw the curved notch edge and bar outline.
    canvas.drawPath(
      barPath,
      Paint()
        ..color = const Color(0x554B5A70)
        ..style = PaintingStyle.stroke
        ..strokeWidth = 1.1
        ..isAntiAlias = true,
    );
  }

  @override
  bool shouldRepaint(covariant _NavbarPainter oldDelegate) {
    return oldDelegate.centerX != centerX;
  }
}

String prettyDate(DateTime value) {
  const months = [
    'Jan',
    'Feb',
    'Mar',
    'Apr',
    'May',
    'Jun',
    'Jul',
    'Aug',
    'Sep',
    'Oct',
    'Nov',
    'Dec',
  ];

  final hour12 =
      value.hour % 12 == 0 ? 12 : value.hour % 12;

  final minute =
      value.minute.toString().padLeft(2, '0');

  return '${value.day} '
      '${months[value.month - 1]} '
      '${value.year}  â€¢  '
      '$hour12:$minute '
      '${value.hour < 12 ? 'AM' : 'PM'}';
}

/// A quiet frosted-glass surface, intentionally unlike neon/glossy cards.
class SoftGlassCard extends StatelessWidget {
  const SoftGlassCard({
    super.key,
    required this.child,
    this.padding = const EdgeInsets.all(18),
    this.radius = 26,
    this.light = false,
    this.onTap,
  });

  final Widget child;
  final EdgeInsetsGeometry padding;
  final double radius;
  final bool light;
  final VoidCallback? onTap;

  @override
  Widget build(BuildContext context) {
    final shape = BorderRadius.circular(radius);
    return ClipRRect(
      borderRadius: shape,
      child: BackdropFilter(
        filter: ImageFilter.blur(sigmaX: 17, sigmaY: 17),
        child: DecoratedBox(
          decoration: BoxDecoration(
            borderRadius: shape,
            color: light
                ? const Color(0xC8E4EDF3)
                : const Color(0x3C7D97AE),
            border: Border.all(
              color: Colors.white.withValues(alpha: light ? .75 : .23),
              width: 1,
            ),
            boxShadow: [
              BoxShadow(
                color: const Color(0xFF071225).withValues(alpha: .13),
                blurRadius: 22,
                offset: const Offset(0, 10),
              ),
            ],
          ),
          child: Material(
            color: Colors.transparent,
            child: onTap == null
                ? Padding(padding: padding, child: child)
                : InkWell(
                    onTap: onTap,
                    borderRadius: shape,
                    child: Padding(padding: padding, child: child),
                  ),
          ),
        ),
      ),
    );
  }
}

class _ArcPainter extends CustomPainter {
  _ArcPainter({required this.ratio});
  final double ratio;
  static const _start = math.pi;
  static const _sweep = math.pi;

  @override
  void paint(Canvas canvas, Size size) {
    final center = Offset(size.width / 2, size.height - 7);
    final radius = math.min(size.width * .405, size.height * .81);
    final rect = Rect.fromCircle(center: center, radius: radius);

    final track = Paint()
      ..color = const Color(0x556980A2)
      ..strokeCap = StrokeCap.round
      ..style = PaintingStyle.stroke
      ..strokeWidth = 16;
    canvas.drawArc(rect, _start, _sweep, false, track);

    final ticks = Paint()
      ..color = const Color(0x5BD7E6F3)
      ..strokeWidth = 1.6
      ..strokeCap = StrokeCap.round;
    for (var i = 0; i <= 36; i++) {
      final a = _start + _sweep * i / 36;
      final inner = radius + 18.0;
      final outer = inner + ((i % 6 == 0) ? 9 : 5);
      final unit = Offset(math.cos(a), math.sin(a));
      canvas.drawLine(center + unit * inner, center + unit * outer, ticks);
    }

    if (ratio > 0) {
      final progress = Paint()
        ..style = PaintingStyle.stroke
        ..strokeWidth = 16
        ..strokeCap = StrokeCap.round
        ..shader = const SweepGradient(
          startAngle: math.pi,
          endAngle: math.pi * 2,
          tileMode: TileMode.clamp,
          colors: [Color(0xFF76B7FA), Color(0xFFB4EBFF)],
        ).createShader(rect);
      canvas.drawArc(rect, _start, _sweep * ratio, false, progress);
    }
  }

  @override
  bool shouldRepaint(covariant _ArcPainter oldDelegate) => oldDelegate.ratio != ratio;
}
